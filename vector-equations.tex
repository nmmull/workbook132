\begin{tasks}[resume, style=enumerate](1)
\task
  For any two vectors $\vv_1$ and $\vv_2$ in $\R^n$, there is a
  vector $\vu$ such that $\vv_1 + \mathbf v_2 + \vu =
  \vzero$.
\task
  For any vectors $\vv_1$, $\vv_2$, and $\vv_3$ in $\R^n$, if $\vv_1
  \in \vspan\{\vv_2, \vv_3\}$, then $\vv_2 \in \vspan\{\vv_1,
  \vv_3\}$.
\task
  The span of any two distinct nonzero vectors in $\R^3$ is a plane.
\task

  $\vspan\{\vv_1, \vv_2, \vv_3\} = \vspan\{\vv_1 + \vv_3, \vv_2\}$.
\end{tasks}

\section{More Difficult Problems}

\begin{tasks}[resume, style=enumerate](1)
\task
  Consider the vectors
  \begin{displaymath}
    \vv_1 = \begin{bmatrix} 1 \\ 1 \\ 1 \\ 1 \end{bmatrix}
    \qquad
    \vv_2 = \begin{bmatrix} 1 \\ 1 \\ 2 \\ 2 \end{bmatrix}
  \end{displaymath}
  Write vectors $\vv_3$ and $\vv_4$ such that $\vv_3$ is not in
  $\vspan\{\vv_1, \vv_2\}$ and $\vv_4$ is not in $\vspan\{\vv_1,
  \vv_2, \vv_3\}$.
\task
  Determine a vector with integer entries that appears in both of the
  following spans.
  \begin{displaymath}
    \vspan\left\{
    \vThree 1 2 {-1},
    \vThree 0 1 1
    \right\}
    \qquad
    \vspan\left\{
    \vThree 1 1 1,
    \vThree 0 1 3
    \right\}
  \end{displaymath}
\task
  Repeat the previous problem with the following spans.
  \begin{displaymath}
    \vspan\left\{
    \vThree 1 0 0,
    \vThree 0 1 0
    \right\}
    \qquad
    \vspan\left\{
    \vThree 1 2 {-1},
    \vThree 0 3 2
    \right\}
  \end{displaymath}
\task
  Let $\vv_1, \vv_2, \vv_3, \vv_4, \vv_5, \vv_6$ be vectors such that
  the matrix $[\vv_1 \ \ \vv_2 \ \ \vv_3 \ \ \vv_4 \ \ \vv_5 \ \ \vv_6]$
  is row equivalent to the following matrix.
  \begin{displaymath}
    \begin{bmatrix}
      1 & -3 & -11 & -2 & 9 & 1 \\
      0 & -2 & -8 & -4 & 6 & -5 \\
      0 & 0 & 4 & 4 & -2 & 1 \\
      0 & 0 & 0 & 0 & 8 & 5
    \end{bmatrix}
  \end{displaymath}
  Determine the first vector (i.e., the vector with the smallest
  index) which is in the span of the vectors which precede it.
  Express this vector as a linear combination of the vectors which
  precede it.  That is, determine the vectors $\vv_i$ with the
  smallest index $i$ such that $\vv_i \in \vspan \{ \vv_1, \dots,
  \vv_{i - 1}\}$ and write $\vv_i$ as a linear combination of the
  vectors $\vv_1, \dots, \vv_{i - 1}$.
\end{tasks}

\section{Challenge Problems}

\begin{tasks}[resume, style=enumerate](1)
\task
  Determine a linear equation in three variables whose point set is exactly
  \begin{align*}
    \left\{
    \vv +
    \begin{bmatrix} 2 \\ -3 \\ 2 \end{bmatrix}
    \ \text{where} \
    \vv \in
    \vspan
    \left\{
    \begin{bmatrix} 1 \\ 1 \\ -5 \end{bmatrix},
    \begin{bmatrix} 0 \\ 1 \\ -1 \end{bmatrix}
    \right\}
    \right\}
  \end{align*}
  In other words, every point in the point set of the equation can be
  expressed as the sum of $[2 \ \ (-3) \ \ 2]^T$ and a vector in the
  span of $[1 \ \ 1 \ \ (-5)]^T$ and $[0 \ \ 1 \ \ (-1)]^T$.
\task
  Consider the following linear equation.
  \begin{displaymath}
    x + y + z = 5
  \end{displaymath}
  The plane represented by this equation does not include the origin;
  such a plane is called \textit{affine}.  Determine vectors $\vb$,
  $\vv_1$, and $\vv_2$ such that every point in the plane represented
  by the above equation (that is, every point $(a, b, c)$ such that $a
  + b + c = 5$) can be written as $\vb + \vv$ where $\vv$ is in
  $\vspan \{\vv_1, \vv_2\}$, i.e.,
  \begin{displaymath}
    \{(x, y, z) : x + y + z = 5\}
    = \{\vb + \vv : \vv \in \vspan\{\vv_1, \vv_2\}\}.
  \end{displaymath}
\end{tasks}

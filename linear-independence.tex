dependent.  If they are, write a dependence relation, i.e., determine
a linear combination of the given vectors which sums to $\vzero$.

\section{True/False}

\trueFalseHeader

\begin{tasks}[resume, style=enumerate](1)
\task
  For $A \in \R^{m \times n}$, if the columns of $A$ are linearly
  dependent, then they do not span $\R^m$.
\task
  If the columns of $A$ are linearly independent, then the reduced
  echelon form of $A$ has only $0$s and $1$s.
\task
  For any vectors $\vv_1$, $\vv_2$, $\vv_3$, if $\vv_1 \in
  \vspan\{\vv_2, \vv_3\}$ then $\{ \vv_1 + \vv_2, \vv_1 + \vv_3\}$ is
  a linearly dependent set.
\task
  For any $m \times n$ matrix $A$, if $m > n$ then the columns of $A$
  must be linearly independent.
\task
  For any vectors $\vv_1$, $\vv_2$, and $\vv_3$ in $\R^n$, if
  $\{\vv_1, \vv_2, \vv_3\}$ is linearly independent, then so is
  $\{\vv_1, \vv_2\}$.
\task
  For any matrix $A$ in $\R^{m \times n}$ and distinct vectors
  $\vv_1$ and $\vv_2$ in $\R^n$, if $A\vv_1 = \vzero$ and
  $A\vv_2 = \vzero$, then the columns of $A$ are linearly dependent.
\task
  For any matrix $A$ in $\R^{m \times n}$, if the columns of $A^T$ are
  linearly dependent, then the columns of $A$ do not span $\R^m$.
\task
 If the vectors $\vv_1$, $\vv_2$, $\vv_3$ are linearly dependent, then
 $\vv_3 \in \vspan\{\vv_1, \vv_2\}$.
\task
  For any vectors $\vv_1, \vv_2, \vv_3 \in \R^n$, if $\{\vv_1, \vv_2,
  \vv_3\}$ is linearly dependent, then so is the set $\{\vv_1 + \vv_2,
  \vv_2 + \vv_2, \vv_3 + \vv_2\}$.
\end{tasks}

\section{More Difficult Problems}

\begin{tasks}[resume, style=enumerate](1)
\task
  Consider four vectors $\vv_1$, $\vv_2$, $\vv_3$ and $\vv_4$ in
  $\R^4$ such that
  \begin{align*}
    \begin{bmatrix}
      \vv_1 & \vv_2 & \vv_3 & \vv_4
    \end{bmatrix}
    \sim
    \begin{bmatrix}
      1 & 1 & 3 & -2 \\
      0 & 4 & -4 & 12 \\
      0 & 0 & 3 & 3 \\
      0 & 0 & 0 & 0
    \end{bmatrix}
  \end{align*}
  \begin{enumerate}[label=(\alph*)]
  \item
    Determine if $\vv_4 \in \vspan\{\vv_1, \vv_2, \vv_3\}$.  If so,
    express $\vv_4$ as a linear combination of the vectors $\vv_1$,
    $\vv_2$ and $\vv_3$.
  \item Determine if vectors $\vv_2$, $\vv_3$, and $\vv_4$ are linearly
    independent.  If so, justify your answer.  If not, determine a
    dependence relation for these vectors.
  \end{enumerate}
\task
  Determine all values of $h$ for which the following set of vectors
  is linearly dependent.
  \begin{align*}
    \vv_1 =
    \begin{bmatrix} 1 \\ 9 \\ -3 \end{bmatrix}
    \qquad
    \vv_2 =
    \begin{bmatrix} 1 \\ 4 \\ 3 \end{bmatrix}
    \qquad
    \vv_3 =
    \begin{bmatrix} -2 \\ h \\ -6 \end{bmatrix}
    \qquad
  \end{align*}
\task
  Determine three \textit{nonzero} vectors $\vv_1$, $\vv_2$, and
  $\vv_3$ in $\R^3$ such that
  \begin{itemize}
  \item
    $\{\vv_1, \vv_2, \vv_3\}$ is linearly dependent and
  \item
    $\vv_1$ cannot be written as a linear combination of $\vv_2$ and
    $\vv_3$.
  \end{itemize}
\task
  Consider the following matrix $A$ and vector $\vb$.
  \begin{displaymath}
    A =
    \begin{bmatrix}
      1 & 1 & 5 & 1 \\
      -1 & 0 & -3 & 2 \\
      2 & 3 & 12 & 4 \\
    \end{bmatrix}
    =
    \begin{bmatrix}
      \va_1 & \va_2 & \va_3 & \va_4
    \end{bmatrix}
    \qquad
    \vb =
    \begin{bmatrix}
      5 \\ -3 \\ 11
    \end{bmatrix}
  \end{displaymath}
  \begin{enumerate}[label=(\alph*)]
  \item
    Determine if $\vb \in \col A$, i.e., if the vector $\vb$ is in the
    span of the columns of $A$. If it is, then express $\vb$ as a
    nontrivial linear combination of the columns of $A$ (you should
    use $\va_i$ to denote the $i^{\text{th}}$ column of $A$). If it
    is not, then justify your answer.
  \item
    Determine if the set of vectors $\{ \va_1, \va_2, \va_4 \}$ is
    linearly dependent.  If it is, write a dependence relation for
    this set of vectors. If it is not, then justify your answer.
  \end{enumerate}

  \item Determine if the columns of this matrix are linearly

  \begin{align*}

    \vv_2 =
    \begin{bmatrix} -3 \\ 3 \\ -8 \\ 5 \\ -2 \\ 7 \\ -3 \end{bmatrix}

  \begin{align*}

    \begin{bmatrix} -2 \\ -10 \\ 10 \\ -5 \\ -3 \\ -10 \\ -5 \end{bmatrix}

    \begin{itemize}

\task
  Consider three arbitrary vectors in $\R^3$.
  \begin{displaymath}
    \vv = \vThree {v_1} {v_2} {v_3}
    \qquad
    \vw = \vThree {w_1} {w_2} {w_3}
    \qquad
    \vu = \vThree {u_1} {u_2} {u_3}
  \end{displaymath}
  and suppose that they are linearly independent.
  \begin{enumerate}[label=(\alph*)]
  \item Determine the maximum number of entries of these vectors which
    can be $0$.
  \item Determine the minimum number of entries which can be $0$.
  \end{enumerate}
  Provide an example in each case.
\task
  Suppose that $\{\vv_1, \vv_2, \vv_3\}$ is a linearly independent set
  of vectors and that
  \begin{align*}
    \vu_1 &= \vv_1 + \vv_3 \\
    \vu_2 &= -2 \vv_1 + \vv_2 + \vv_3 \\
    \vu_3 &= -3 \vv_1 - \vv_2 - 6 \vv_3
  \end{align*}
  Determine a dependence relation with integer weights for the vectors
  $\{\vu_1, \vu_2, \vu_3\}$.
\task
  Consider the following vectors in $\R^4$.
  \begin{displaymath}
    \vv_1 = \vFour {-1} {-6} 0 7
    \qquad
    \vv_2 = \vFour 1 {-2} {-3} {-1}
    \qquad
    \vv_3 = \vFour 1 {-1} {-2} 0
    \qquad
    \vv_4 = \vFour 0 {-1} 0 {-1}
  \end{displaymath}
  \begin{enumerate}[label=(\alph*)]
  \item
    Determine if $\vv_1$ is in $\vspan\{\vv_2, \vv_3, \vv_4\}$.
    Justify your answer.  In particular, if $\vv_1$ is in
    $\vspan\{\vv_2, \vv_3, \vv_4\}$, then write $\vv_1$ as a linear
    combination of $\vv_2$, $\vv_3$, and $\vv_4$.
  \item
    Determine if the vectors $\vv_2$, $\vv_3$, and $\vv_4$ are
    linearly independent.  Justify your answer.  In particular, if
    they are linearly dependent, then write a dependence relation for
    them (that is, write the zero vector $\vzero$ as a linear
    combination of the vectors $\vv_2$, $\vv_3$, and $\vv_4$).
  \item
    Determine if the vectors $\vv_1$, $\vv_2$, and $\vv_3$ are
    linearly independent.  Justify your answer.  In particular, if
    they are linearly dependent, then write a dependence relation for
    them.
  \end{enumerate}

    \qquad
    \vv_3 = \vFour 0 {-2} {-1} {-1}
    \qquad
    \vv_4 = \vFour 1 1 1 {-2}

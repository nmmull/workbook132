\section{True/False}

\trueFalseHeader

\begin{tasks}[resume, style=enumerate](1)
\task
  There is a unique basis for any subspace.
\task
  A $7 \times 4$ matrix $A$ may have $\dim(\nul A) = 5$.
\task
  A $3 \times 6$ matrix $A$ may have $\dim(\nul A) = 2$.
\task
  For any matrix $A \in \R^{n \times n}$, if $A$ is invertible then
  $\rank A = n$.
\task
  For any matrix $A \in \R^{m \times n}$, $\col(A)$ is the same as the
  set of vectors $\vb$ such that $A \vx = \vb$ has a solution.
\task
  For any matrix $A$, we have $\rank(A) = \rank(A^T)$.
\task
  A basis is a spanning set that is as large as possible.
\task
  A linear transformation $T: \R^3 \to \R^3$ that maps $\R^3$ to a
  plane has a trivial kernel (i.e., if $T(\vv) = \vzero$, then $\vv =
  \vzero$).
\task
  If $\cB$ is the standard basis for $R^n$, then for any $\vx \in
  \R^n$ we have that $[\vx]_{\cB} = \vx$
\task
  For any matrix $A$, the dimension of the null space of $A$ is at
  most the rank of $A$.
\end{tasks}

\section{More Difficult Problems}

\begin{tasks}[resume, style=enumerate](1)
\task
  For a matrix $A \in \R^{m \times n}$, consider the set of vectors
  $\{\vx \in \R^n : A\vx= \ve_1\}$ (\textit{Recall:} $\ve_1$ is the
  first standard basis vector).
  \begin{enumerate}
  \item
    Determine if this set if closed under addition. Justify your
    answer.
  \item
    Determine if this set is closed under scaling. Justify your
    answer.
  \item
    Determine if this set is a subspace of $\R^n$. Justify your
    answer.
  \end{enumerate}
\task
  \begin{align*}
    \vv_1 = \begin{bmatrix}
      3  \\
      0  \\
      -1
    \end{bmatrix},
    \vv_2 = \begin{bmatrix}
      -3  \\
      2  \\
      0
    \end{bmatrix},
    \vv_3 = \begin{bmatrix}
      0  \\
      -2  \\
      -1
    \end{bmatrix}
\end{align*}
List all possible subsets of the above vectors that form a basis of
the subspace $\vspan\{\vv_1, \vv_2, \vv_3\}$.\footnote{The process we
teach for determining a basis gives one choice, but there may be
multiple choices that are suitable.}
\task Consider the following vectors.
\begin{align*}
\vv_1 = \begin{bmatrix}
      3   \\
      -1
\end{bmatrix},
\vv_2 = \begin{bmatrix}
      -3  \\
      1
\end{bmatrix},
\vv_3 = \begin{bmatrix}
      -1  \\
      1
\end{bmatrix},
\vv_4 = \begin{bmatrix}
      8  \\
      -4
\end{bmatrix}
\end{align*}
List all possible subsets of the above vectors that form a basis of
the subspace $\vspan\left\{\vv_1,  \vv_2,
\vv_3,  \vv_4 \right\}$.
\task Consider the 3-dimensional vector space of all quadratic
  polynomials $Q = \{a x^2 + b x + c  |  a,b,c \in \R\}$
  and consider the linear derivative map $\frac{d}{dx}: Q \to Q$
  defined by $\frac{d}{dx}(a x^2 + b x + c) = 0 x^2 + 2 a x + b$.
  \begin{enumerate}
  \item
    Using the standard basis given by $\{x^2, x, 1 \}$, determine a $3
    \times 3$ matrix $A$ that implements $\frac{d}{dx}$.
  \item
    Determine a basis for $\col A$.
  \item
    Determine basis for $\nul A$.
  \item
    Determine $\rank A$ and $\dim(\nul A)$.
  \end{enumerate}
\task
  Let $A$ be an $m \times n$ matrix and let $\vv$ be a vector in
  $\R^n$.  Let $\vb$ denote the vector $A\vv$.
  \begin{enumerate}
  \item
    Show that if $\vw \in \nul A$, then $\vv + \vw$ is a solution to
    the equation $A\vx = \vb$.
  \item
    Show that if $\vb \not = 0$, then the solution set of $A\vx = \vb$
    is not a subspace of $\R^n$.
  \item
    A set $H$ is an \textbf{affine} subspace of $\R^n$ if there is a
    subspace $U$ of $\R^n$ and a vector $\vo$ such that
    \begin{align*}
      H = \{\vu + \vo \ | \ \vu \in U\}
    \end{align*}
    Show that the solution set of $A\vx = \vb$ is an affine subspace
    of $\R^n$ if it is nonempty.  In particular, choose a vector $\vo$
    and subspace $U$.
  \end{enumerate}
\task
  Consider the following matrix.
  \begin{displaymath}
    A =
    \begin{bmatrix}
      1 & 2 & 3 & -1 \\
      1 & 2 & 4 & -4 \\
      0 & 0 & 2 & 4 \\
      0 & 0 & 1 & 2
    \end{bmatrix}
  \end{displaymath}
  \begin{enumerate}
  \item
    Determine a basis for $\nul A$.
  \item
    Determine a linear equation over the variables $x_1, x_2, x_3,
    x_4$ whose solution set is $\col A$ (that is, $(x_1, x_2, x_3,
    x_4)$ satisfies the linear equation if and only if the vector
    $[x_1 \ x_2 \ x_3 \ x_4]^T$ is in the column space of $A$).
  \end{enumerate}
\task
  Determine the linear equation whose solution set is the column space
  of the following matrix.
  \begin{displaymath}
    \begin{bmatrix}
      1 & -4 & -10 \\
      -4 & 17 & 42 \\
      1 & -2 & -6
    \end{bmatrix}
  \end{displaymath}
\task
  Let $A$ be a $4 \times 2024$ matrix where $\rank A = 3$. Further
  suppose that the LU-decomposition of $A$ has
  \begin{displaymath}
    L =
    \begin{bmatrix}
      1 & 0 & 0 & 0 \\
      -2 & 1 & 0 & 0 \\
      7 & 5 & 1 & 0 \\
      0 & 0 & 1 & 1
    \end{bmatrix}
  \end{displaymath}
  Determine the linear equation whose solution set is $\col A$.
\task
  Consider the following two matrices.
  \begin{displaymath}
    A =
    \begin{bmatrix}
      6 & -6 & -1 & 17 \\
      9 & -2 & 2 & 36 \\
      -10 & -13 & 8 & -45 \\
      -9 & 4 & -1 & -33 \\
      2 & 4 & 2 & 14
    \end{bmatrix}
    \qquad
    B =
    \begin{bmatrix}
      -8 & 5 & -16 & 1 \\
      2 & 11 & 4 & 16 \\
      3 & -2 & 6 & 3 \\
      2 & -10 & 4 & 0 \\
      8 & 4 & 16 & 9
    \end{bmatrix}
  \end{displaymath}
  Also consider the following set of vectors.
  \begin{displaymath}
    H = \{ \vu + \vv \ | \ \vu \in \col A \text{ and } \vv \in \col B \}
  \end{displaymath}
  That is, $H$ consists of all sums of pairs of vectors, one from
  $\col A$ and one from $\col B$.
  \begin{enumerate}
  \item
    Show that $H$ is a subspace of $\R^5$.
  \item \faCalculator \
    Determine the dimension of $H$.  Justify your answer.  You may use a
    computer but you must determine any matrix you reduce along with its
    reduced echelon form.
  \end{enumerate}
\end{tasks}

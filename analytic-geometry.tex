\section{True/False}

\trueFalseHeader

\begin{tasks}[resume, style=enumerate](1)
\task
  If $\|\vu\|^2 + \|\vv\|^2 = \|\vu + \vv\|^2$, then $\vu$ and $\vv$
  are orthogonal.
\task
  For a square matrix $A$, vectors in $\col A$ are orthogonal to
  vectors in $\nul A$.
\task
  There can be a linear dependence relationship between vectors in an
  orthogonal set.
\task
  For any $A \in \R^{m \times n}$, there is an eigenbasis of $\R^n$
  for $A^TA$.
\task
  In $\R^4$, any set of vectors that has 5 members cannot be an
  orthogonal set.
\task
  Orthogonal matrices are invertible.
\task
  Orthogonal matrices have determinant 1.
\task
  For any matrix $A \in \R^{m \times n}$ and vector $\vv \in \R^n$, if
  $\|A \vv \| = \vzero$ then $\vv \in \nul A$.
\task
  For an $m \times n$ matrix with orthogonal columns, it may be the
  case that $m < n$.
\task
  For any matrix $U \in \mathbb \R^{n \times n}$, if $U$ is orthogonal
  then $\det U = \pm 1$.
\task
  The orthogonal projection of $\vy$ onto $\vv$ is the same as the
  orthogonal projection of $\vy$ onto $c \vv$ whenever $c \neq 0$.
\task
  For any vector $\vy \in \R^n$ and any subspace $W$ of $\R^n$, the
  vector $\vy - \proj_W \vy$ is orthogonal to $W$.
\task
  For any matrix $A \in \R^{n \times n}$, if $A$ has an eigenbasis of
  $\R^n$ then $A$ is symmetric.
\task
  For any matrix $A$ with orthogonal columns, $A^TA = I$.
\task
  Every orthogonal set is linearly independent.
\task
  For any matrix $A$, if $A$ is orthogonally diagonalizable, then so
  is $A^T$.
\task
  $\|A \vv \| = \| A^2 \vv \|$ for any orthogonal matrix $A \in \R^{n
  \times n}$ and any vector $\vv \in \R^n$.
\task
  For any vector $\vu \in \R^n$, if $\|\vu\| = 1$ then the matrix $\vu
  \vu^T$ implements projection onto $\vspan\{\vu\}$.
\end{tasks}

\task
  Consider the following vectors.\footnotemark
